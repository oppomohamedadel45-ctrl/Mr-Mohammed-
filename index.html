using System; using System.Collections.Generic; namespace GlobalWarGame { // فئة تمثل الدولة (اللاعب أو العدو) public class Nation { public string Name { get; set; } public int Money { get; set; } public int Food { get; set; } public int Steel { get; set; } public int Army { get; set; } public int Factories { get; set; } public int Farms { get; set; } public int TechLevel { get; set; } public bool IsAlive { get; set; } public Nation(string name, int money, int food, int steel, int army, int factories, int farms, int tech) { Name = name; Money = money; Food = food; Steel = steel; Army = army; Factories = factories; Farms = farms; TechLevel = tech; IsAlive = true; } public void CollectResources() { if (!IsAlive) return; Money += (Factories * 75) + 50; Food += (Farms * 60) + 40; Steel += (Factories * 45) + 30; } } class Program { static void Main(string[] args) { // محاكي C# الاحترافي للألعاب (يمكن تشغيله في أي مترجم C# أونلاين مثل Programiz أو Replit) Nation player = new Nation("جمهورية مصر العظمى", 600, 300, 200, 200, 2, 2, 1); List enemies = new List { new Nation("الإمبراطورية المجاورة", 700, 350, 300, 220, 3, 2, 1), new Nation("تحالف القوى الكبرى", 800, 400, 400, 280, 4, 3, 2) }; int turn = 1; bool playing = true; Console.WriteLine("=== مرحباً بك في محاكي الحروب العالمي (بأفضل معايير لغة C#) ==="); while (playing && player.IsAlive) { Console.WriteLine($"\n================ الشهر الحالي: {turn} ================"); Console.WriteLine($"[دولتـك]: {player.Name}"); Console.WriteLine($"المال: {player.Money} | المؤن: {player.Food} | الصلب: {player.Steel} | الجيش: {player.Army} جندي"); Console.WriteLine($"المصانع: {player.Factories} | المزارع: {player.Farms} | التكنولوجيا: مستوى {player.TechLevel}"); Console.WriteLine("--------------------------------------------------"); Console.WriteLine("1. بناء مصنع عسكري (150 مال)"); Console.WriteLine("2. إنشاء مزرعة طعام (100 مال)"); Console.WriteLine("3. تموين وإطعام الجنود (40 مؤن)"); Console.WriteLine("4. ابتكار عتاد عسكري متقدم (150 مال + 100 صلب)"); Console.WriteLine("5. شن حرب واكتساح عدو"); Console.WriteLine("6. إنهاء الشهر وجمع الموارد"); Console.Write("اختر رقم العمليات: "); string input = Console.ReadLine(); if (input == "1" && player.Money >= 150) { player.Money -= 150; player.Factories++; Console.WriteLine("-> تم بناء مصنع سلاح جديد بنجاح."); } else if (input == "2" && player.Money >= 100) { player.Money -= 100; player.Farms++; Console.WriteLine("-> تم بناء مزرعة جديدة بنجاح."); } else if (input == "3" && player.Food >= 40) { player.Food -= 40; player.Army += 20; Console.WriteLine("-> تم إطعام الجنود ورفع كفاءة الجيش."); } else if (input == "4" && player.Money >= 150 && player.Steel >= 100) { player.Money -= 150; player.Steel -= 100; player.TechLevel++; Console.WriteLine("-> تم ابتكار وتطوير جيل جديد من العتاد الحربي!"); } else if (input == "5") { Console.WriteLine("\nاختر العدو للهجوم عليه:"); for (int i = 0; i < enemies.Count; i++) { if (enemies[i].IsAlive) Console.WriteLine($"{i + 1}. {enemies[i].Name} (جيش العدو: {enemies[i].Army})"); } int enemyIndex = Convert.ToInt32(Console.ReadLine()) - 1; if (enemyIndex >= 0 && enemyIndex < enemies.Count && enemies[enemyIndex].IsAlive) { Nation target = enemies[enemyIndex]; Console.WriteLine($"\n[!] تتقدم طوابير العتاد والجيوش نحو حدود {target.Name}..."); int playerPower = (player.Army * 1.2) * (player.TechLevel * 0.5); int enemyPower = (target.Army * 1.0) * (target.TechLevel * 0.4); if (playerPower > enemyPower) { Console.WriteLine($"[انتصار ساحق] سحقت قواتنا جيش {target.Name} وغنمنا خزائنهم!"); player.Money += target.Money; player.Steel += target.Steel; target.IsAlive = false; } else { Console.WriteLine($"[هزيمة مريرة] تصدى العدو لهجومنا وكبدنا خسائر فادحة!"); player.Army -= 50; } } } else if (input == "6") { player.CollectResources(); turn++; Console.WriteLine("-> انتهى الشهر. تم جمع أرباح المصانع والمزارع."); } // التحقق من فناء الجيش if (player.Army <= 10 && player.Money < 30) { Console.WriteLine("\n=== سقوط الإمبراطورية: تم تدمير جيشك بالكامل! ==="); break; } } } } }
