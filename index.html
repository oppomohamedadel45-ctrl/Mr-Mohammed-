class SchoolSystem: def __init__(self): # قاعدة بيانات النظام الوهمية self.users = { "super_admin": {"pass": "AdminRoot#2026", "role": "Super Admin", "name": "المنسق العام"}, "TCH-101": {"pass": "Chem@2026_#1", "role": "Teacher Admin", "name": "أستاذ الكيمياء", "students": []}, "TCH-102": {"pass": "Phys@2026_#2", "role": "Teacher Admin", "name": "أستاذ الفيزياء", "students": []}, "TCH-103": {"pass": "Bio@2026_#3", "role": "Teacher Admin", "name": "أستاذ الأحياء", "students": []}, "TCH-104": {"pass": "Arab@2026_#4", "role": "Teacher Admin", "name": "أستاذ اللغة العربية", "students": []}, "TCH-105": {"pass": "Eng@2026_#5", "role": "Teacher Admin", "name": "أستاذ اللغة الإنجليزية", "students": []} } def login(self, username, password): if username in self.users and self.users[username]["pass"] == password: print(f"\n[تسجيل دخول ناجح] أهلاً بك يا {self.users[username]['name']} ({self.users[username]['role']})") return username else: print("\n[خطأ] اسم المستخدم أو كلمة المرور غير صحيحة.") return None def add_student(self, teacher_id, student_name): if teacher_id in self.users and self.users[teacher_id]["role"] == "Teacher Admin": self.users[teacher_id]["students"].append(student_name) print(f"تم إضافة الطالب '{student_name}' بنجاح إلى صفحة الأستاذ: {self.users[teacher_id]['name']}") else: print("غيرحساب مدرس صالح.") def view_dashboard(self, username): user_info = self.users[username] if user_info["role"] == "Super Admin": print("\n--- لوحة تحكم المشرف الأعلى (الكل) ---") for t_id, data in self.users.items(): if data["role"] == "Teacher Admin": print(f"المدرس: {data['name']} (كود: {t_id}) | عدد الطلاب: {len(data['students'])} | الأسماء: {data['students']}") else: print(f"\n--- لوحة تحكم المدرس: {user_info['name']} ---") print(f"قائمة طلابك الخاصة: {user_info['students']}") #
